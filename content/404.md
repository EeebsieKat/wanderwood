<div style="text-align: center; font-family: monospace;">
  <h2>404 - Page Lost in the Wanderwood!</h2>
  <p>The note you're looking for was consumed by the roots. Play a quick game to find your way back!</p>
  
  <canvas id="flappyCanvas" width="320" height="480" style="border:2px solid #565f89; background:#16161e; border-radius:8px; cursor:pointer;"></canvas>
  <p style="font-size: 0.8rem; color: #cfc9c2;">Press <b>SPACE</b> or <b>CLICK</b> to Jump | Refresh to Restart</p>
</div>

<script>
  const canvas = document.getElementById('flappyCanvas');
  const ctx = canvas.getContext('2d');

  let birdY = 200, velocity = 0, gravity = 0.35, jump = -6.5;
  let score = 0, gameOver = false;
  let pipes = [];
  const pipeWidth = 40, pipeGap = 120;

  function spawnPipe() {
    let topHeight = Math.floor(Math.random() * (canvas.height - pipeGap - 80)) + 30;
    pipes.push({ x: canvas.width, top: topHeight, passed: false });
  }

  function update() {
    if (gameOver) return;

    velocity += gravity;
    birdY += velocity;

    if (birdY > canvas.height - 15 || birdY < 0) gameOver = true;

    if (pipes.length === 0 || pipes[pipes.length - 1].x < canvas.width - 160) {
      spawnPipe();
    }

    for (let i = 0; i < pipes.length; i++) {
      let p = pipes[i];
      p.x -= 2;

      // Collision Check
      if (
        20 + 15 > p.x && 20 < p.x + pipeWidth &&
        (birdY < p.top || birdY + 15 > p.top + pipeGap)
      ) {
        gameOver = true;
      }

      if (!p.passed && p.x < 20) {
        p.passed = true;
        score++;
      }
    }

    pipes = pipes.filter(p => p.x > -pipeWidth);
  }

  function draw() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);

    // Draw Player (Glowing Block)
    ctx.fillStyle = '#2ac3de';
    ctx.fillRect(20, birdY, 15, 15);

    // Draw Pipes
    ctx.fillStyle = '#7dcfff';
    pipes.forEach(p => {
      ctx.fillRect(p.x, 0, pipeWidth, p.top);
      ctx.fillRect(p.x, p.top + pipeGap, pipeWidth, canvas.height - (p.top + pipeGap));
    });

    // Draw Score
    ctx.fillStyle = '#cfc9c2';
    ctx.font = '16px monospace';
    ctx.fillText(`Score: ${score}`, 10, 25);

    if (gameOver) {
      ctx.fillStyle = '#bb9af7';
      ctx.font = '20px monospace';
      ctx.fillText('GAME OVER', 100, 230);
      ctx.font = '12px monospace';
      ctx.fillText('Click or Space to Retry', 80, 260);
    }
  }

  function loop() {
    update();
    draw();
    requestAnimationFrame(loop);
  }

  function doJump() {
    if (gameOver) {
      birdY = 200; velocity = 0; score = 0; pipes = []; gameOver = false;
    } else {
      velocity = jump;
    }
  }

  window.addEventListener('keydown', (e) => { if (e.code === 'Space') { e.preventDefault(); doJump(); } });
  canvas.addEventListener('click', doJump);

  loop();
</script>
